
ACTOR Loader_3D : Loader replaces Loader
{
  States
  {
  Ready:
    SAWG A 0 ACS_NamedExecuteAlways("LOADUP", 0, 20, 56)
    SAWG A 1 A_WeaponReady
	loop
  Deselect: 
	SAWG A 0 A_Stopsound(CHAN_WEAPON)
	SAWG A 0 A_SetGravity (1)
	//SAWG A 3 A_Lower
	SAWG A 0 A_SpawnItemEx("LBerserk_3D", 72, 0, 24, 4, 0, 0, 0, SXF_SETMASTER)
	SAWG A 0 A_PlaySound ("SYNTPN2", 1, 1.0)
	SAWG A 0 A_TakeInventory("Loader_3D",1)
	Stop
  Select:
    // 2026-11-08: A_SetGravity(3) and A_SetSpeed(0.35) are gone from here. The speed never did
    // anything (PlayerPawn.MovePlayer rewrites Speed every tic; AE_LoaderWorn.GetSpeedFactor is
    // the slowdown) and the gravity DID: a triple-speed fall is vertical acceleration the VR
    // player's body never made. AE_LoaderWorn owns gravity now (ae_loader_gravity, 1.0).
    SAWG A 1 A_Raise
	Loop
  Fire:
	// 2026-11-08: THE DECORATE NO LONGER ATTACKS. Until now a trigger press ran the TC's own
	// A_CustomPunch(500) twice from the view line AND closed the claw; alt fire ran three
	// A_Explode(20, 256) AND the ram. The arms are AE_LoaderHandler's (zscript/ae/loader.zs):
	// claw / carry / crush / fling / ram on the main arm, arc / torch on the off arm, the swing
	// on both, all at the CLAW's point, all through net events. A desktop (non-VR) player has no
	// claw point and so no attack on this weapon; this is the VR pack. The old states are in
	// git (1821273) if a desktop fallback is ever wanted.
	SAWG A 1
	Goto Ready
  AltFire:
	SAWG A 1
	Goto Ready
  AltFlash:	
	SAWS ABCDEFGHIJ 1 Bright A_Light2
	SAWS ABCDEFGHIJ 1 Bright A_Light2
	TNT1 A 2 A_Light1
	TNT1 A 0 A_Light0
	Stop
  }
}

ACTOR LBerserk_3D : LBerserk replaces LBerserk
{
	States
	{
	Spawn:
		CSAW A 0 A_JumpIf(GetCVAR("avw_pickupmodel")==1, "Spawn3D")
		CSAW A 1
		Loop
	Spawn3D:
		CSAW A 0 A_JumpIf(GetCVAR("avw_pickupmodel")==0, "Spawn")
		CSAW P 1
		Loop
	Pickup:
		TNT1 A 0 A_GiveInventory("Loader_3D")
		TNT1 A 0 A_SelectWeapon("Loader_3D")
		Stop
	}
}
